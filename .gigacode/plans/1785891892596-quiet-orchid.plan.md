# Модернизация системы хранения данных смен

## Цель
Заменить хранение данных смен в SharedPreferences на файловую систему. Каждая смена = один JSON-файл. SharedPreferences используется только для текущих данных и настроек.

## Формат файла смены
**Путь:** `/Download/Ohrana/Shifts/shift_{shiftId}_{date}.json`

**Пример:** `/Download/Ohrana/Shifts/shift_NS060826_001_20260806.json`

**Структура JSON:**
```json
{
  "shiftId": "NS060826_001",
  "startTime": "06.08.2026 14:00:00",
  "endTime": "06.08.2026 22:00:00",
  "guards": [{"nfcId": "...", "name": "...", "role": "..."}],
  "rounds": [{"id": 1, "shiftId": "...", "startTime": "...", "endTime": "...", "isCompleted": true, ...}],
  "logs": [{"id": "log_XXX", "shiftId": "...", "roundId": 1, "checkpointName": "...", ...}],
  "violations": [{"id": "violation_XXX", "shiftId": "...", ...}],
  "incidents": [{"id": "incident_XXX", "shiftId": "...", ...}],
  "staffEvents": [{"id": "...", "timestamp": "...", "staffId": "...", ...}],
  "guardMessages": [{"id": "...", "shiftId": "...", "timestamp": "...", ...}],
  "stats": {"completedRounds": 3, "totalViolations": 2, "totalIncidents": 1}
}
```

content = '''package com.example.ohrana

import android.content.Context
import android.util.Log
import org.json.JSONArray
import org.json.JSONObject
import java.io.File
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

data class ShiftFileInfo(
    val filePath: String,
    val fileName: String,
    val shiftId: String,
    val date: String?,
    val time: String?,
    val lastModified: Long
)

class ShiftFileSt(private val context: Context) {

    companion object {
        private const val TAG = "ShiftFileSt"
        private val dateFormat = SimpleDateFormat("dd.MM.yyyy HH:mm:ss", Locale.US)
        private val dateShortFormat = SimpleDateFormat("ddMMyyyy", Locale.US)
    }

    private fun getShiftDirectory(context: Context): File {
        val downloadDir = context.getExternalFilesDir(null)
            ?: return File(context.getExternalFilesDir(android.os.Environment.DIRECTORY_DOWNLOADS) ?: return emptyList()

        return shiftDir.listFiles()?.filter { it.name.startsWith("shift_") && it.name.endsWith(".json") }
            ?.map { file ->
                val shiftInfo = parseDateFromFileName(file.name)
                ShiftFileInfo(
                    filePath = file.absolutePath,
                    fileName = file.name,
                    shiftId = extractShiftIdFromName(file.name),
                    date = shiftInfo?.first,
                    time = shiftInfo?.second,
                    lastModified = file.lastModified()
                )
            }?.sortedByDescending { it.lastModified } ?: emptyList()
    }

    private fun extractShiftIdFromName(fileName: String): String {
        return fileName.removePrefix("shift_")
            .removeSuffix(".json")
            .substringBeforeLast("_")
    }

    private fun parseDateFromFileName(fileName: String): Pair<String, String>? {
        val dateStr = fileName.removePrefix("shift_")
            .removeSuffix(".json")
            .substringAfterLast("_")

        return try {
            val date = SimpleDateFormat("yyyyMMdd", Locale.US).parse(dateStr)
            if (date != null) {
                val dateString = dateFormat.format(date).substring(0, 10)
                val fullTimeString = dateFormat.format(date)
                Pair(dateString, fullTimeString)
            } else {
                null
            }
        } catch (e: Exception) {
            Log.w(TAG, "Could not parse date from filename: $fileName, dateStr=$dateStr")
            null
        }
    }

    fun cleanupOldShiftFiles(daysToKeep: Int, context: Context): Int {
        val cutoffDate = Date(System.currentTimeMillis() - daysToKeep * 24L * 60 * 60 * 1000)
        val shiftDir = getShiftDirectory(context)
        var deletedCount = 0

        shiftDir.listFiles()?.filter { it.name.startsWith("shift_") && it.name.endsWith(".json") }
            ?.forEach { file ->
                val fileDate = parseDateFromFileName(file.name)?.first
                if (fileDate != null) {
                    try {
                        val parsedDate = dateFormat.parse(fileDate)?.time ?: 0L
                        if (parsedDate < cutoffDate.time) {
                            file.delete()
                            deletedCount++
                            Log.d(TAG, "Deleted old shift file: ${file.name}")
                        }
                    } catch (e: Exception) {
                        Log.w(TAG, "Could not parse date for cleanup: ${file.name}")
                    }
                }
            }

        Log.d(TAG, "Cleanup completed: $deletedCount files deleted")
        return deletedCount
    }
}
'''

### Этап 6: Очистка старых файлов смен (опционально)
**Файл:** `app/src/main/java/com/example/ohrana/ShiftFileSt.kt`

```kotlin
fun cleanupOldShiftFiles(daysToKeep: Int, context: Context): Int {
    val cutoffDate = Calendar.getInstance().apply { add(Calendar.DAY_OF_YEAR, -daysToKeep) }
    val shiftDir = getShiftDirectory(context)
    var deletedCount = 0
    
    shiftDir.listFiles()?.filter { it.name.startsWith("shift_") && it.name.endsWith(".json") }?.forEach { file ->
        val fileDate = parseDateFromFileName(file.name)
        if (fileDate != null && fileDate.before(Date(cutoffTime))) {
            file.delete()
            deletedCount++
        }
    }
    return deletedCount
}
```

**Логика очистки:**
- Оставляем файлы за последние 3 дня
- Удаляем более старые файлы
- Можно вызвать из AdministratorScreen вручную

## Совместимость
- Старые данные из SharedPreferences автоматически конвертируются при закрытии
- Файлы уже закрытых смен не создаются (только новые)
- Fallback: если файла нет — читаем из SharedPreferences (для старых смен)
